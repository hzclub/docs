## @reown/appkit/vue

- `@reown/appkit/vue` 是 `@reown/appkit` 生态中针对 Vue.js 框架的官方集成模块，主要用于简化 Web3 应用开发中的钱包连接、多链交互等场景

```js
import { useAppKit, useAppKitAccount, useDisconnect } from "@reown/appkit/vue";
const { disconnect } = useDisconnect();
disconnect()	// 断开连接操作

const { open } = useAppKit();// 启动钱包操作
open({ view: "Connect" });	// 触发钱包连接界面
open({ view: "Account" })	// 打开账户管理界面

const accountData = useAppKitAccount();	// accountData 提供的响应式钱包账户状态对象,核心内容如下
interface AccountData {
  address: string | null;      // 当前连接的区块链地址（如 "0x123..."）
  chainId: number | null;      // 链ID（如以太坊主网为 1）
  isConnected: boolean;        // 钱包是否已授权连接
  connectorType?: string;      // 钱包类型（如 "metamask"、"phantom"）
  balance?: string;            // 可选：账户原生代币余额
}
```

## 登录

```js
<script setup>
    import { onMounted, ref } from 'vue';
    import { watch, computed } from 'vue';
    import { useI18n } from 'vue-i18n';   // 导入国际化模块
    import { useUserStore } from "@/store/user.js";
    import { useAppKit, useAppKitAccount, useDisconnect } from "@reown/appkit/vue";	// 钱包工具
    import { useSignMessage } from "@wagmi/vue";	// 签名
    import { login, check } from '@/api/login';
    import { onLoad } from '@dcloudio/uni-app';
    import { useIndex } from "@/hooks/index.js";
    
    const message = 'PET'	// 签名内容
	const invite_code = ref('')		// 地址栏邀请码
    
    onLoad((options) => {
        invite_code.value = options.invite_code		// 拿到地址栏邀请码
    })
    
	// 连接钱包
    async function handConnect() {
        if (!userStore.isLogin) {
            await disconnect();
        }
        if (accountData.value.isConnected) {	// 是否授权钱包连接
            open({ view: "Account" });	// 已授权连接打开钱包管理页面
        } else {
            open({ view: "Connect" });	// 未授权连接打开钱包登录页面
        }
    }
    // 监听登录状态，监听连接状态
    watch(
        () => accountData.value.isConnected, (val) => {
            userStore.setAddress(accountData.value.address);	// 已连接保存钱包地址
            if (!val) {
                // 钱包未连接清空状态
                userStore.reset()
                console.log('钱包未连接', val)
                return
            }
            if (userStore.isLogin) {
                // 已用户登录不做操作
                return
            }
            // 钱包连接触发签名
            handleSign()
        }
    )
    // 签名
    async function handleSign() {
        uni.showLoading({
            title: t('customize-6'),	// 正在前往签名loading
            mask: true
        })
        try {
            // signMessageAsync 触发签名方法，返回签名信息
            signature = await signMessageAsync({
                message,  // 签名内容，比如 'PET'
                account: accountData.value.address   // 钱包地址
            })
            // 签名成功后检查是否新用户
            invitationCheck()
        } catch (error) {
            disconnect()  // 连接钱包失败，断开钱包
            uni.showToast({
                title: t('customize-7'),
                icon: 'none'
            })
        } finally {
            uni.hideLoading()
        }
    }

    // 新用户验证
    async function invitationCheck() {
        // 传入钱包地址后端返回是否为新用户
        const res = await check(accountData.value.address)	
        // 老用户直接登录
        if (!res.data) {
            loginApi()
            return
        }
        // 新用户且地址栏带邀请码
        if (invite_code.value) {
            loginApi(invite_code.value)
            return
        }
        // 是新用户弹出邀请码弹框
        if (res.data) {
            invitationShow.value = true
            return
        }
    }
    // 登录
    async function loginApi(invite_code = '') {
        try {
            const res = await login({
                address: accountData.value.address,	// 钱包地址
                signature,	// 签名返回签名信息
                message,	// 签名内容
                invite_code: invite_code	// 邀请码，没有邀请码为 ''
            })
            invitationShow.value = false	// 关闭邀请码弹框(新用户才会触发)
            userStore.setToken(res.data.token);	// 保存token
            await getIndexData()	// 获取首页数据
            uni.showToast({
                title: t('customize-8'),	// 登录成功
                icon: "none",
            });
            uni.$emit('login')	// 通知刷新页面数据
        } catch (error) {
            console.log(error)
            uni.showToast({
                title: t('customize-9'),	// 有验证码且错误
                icon: "none",
            });
            invitationShow.value = true	// 弹出邀请码弹窗
        }
    }
    // 取消填写邀请码
    function invitationCancel() {
        disconnect();	// 断开连接
        userStore.reset() // 重置状态
        invitationShow.value = false	// 关闭弹框
    }
	// 页面挂载好进行事件监听
    onMounted(() => {
        // 重新登录
        uni.$on('logout', async () => {
            if (!userStore.isLogin) {
                await disconnect();	// 断开连接
                open({ view: "Connect" });	// 弹出钱包登录
            }
        })
    })
</script>
```

```js
        },
        // 确保加载顺序
```

# 拿取地址栏参数

- 钱包有时候会出现洗参数的情况，需要特定方法拿到地址栏参数

  ```js
  /**
   * 从URL中获取参数
   * @param str 字段名称
   * @returns 参数值
   */
  export function getDataFromUrl(str) {
    // 仅在浏览器环境中执行
    if (typeof window !== 'undefined' && window.location) {
      try {
        const url = window.location.href // 获取当前页面的完整 URL
  
        // 解析完整的 URL
        const urlObj = new URL(url)
  
        // 首先尝试从标准查询参数中获取
        let value = urlObj.searchParams.get(str)
        if (value !== null) {
          return value
        }
  
        // 如果在标准查询参数中没有找到，检查 hash 后面是否有查询参数
        // 例如: http://localhost:9002/?inviteCode=888888#/ 或 http://localhost:9002/#/?inviteCode=888888
        const hash = urlObj.hash
        if (hash.includes('?')) {
          // 提取 hash 中的查询字符串部分，例如从 "#/?inviteCode=888888" 中提取 "?inviteCode=888888"
          const hashQuery = hash.substring(hash.indexOf('?'))
          const hashUrl = new URL(`http://dummy.com${hashQuery}`)
          value = hashUrl.searchParams.get(str)
          if (value !== null) {
            return value
          }
        }
  
        // 如果在 hash 中也没找到，尝试解析整个 URL 字符串中的查询参数
        // 这可以处理类似 http://localhost:9002/?inviteCode=888888#/ 的情况
        const fullSearch = url.split('?')[1]
        if (fullSearch) {
          const searchParts = fullSearch.split('#')
          if (searchParts[0]) {
            const searchParams = new URLSearchParams(searchParts[0])
            value = searchParams.get(str)
            if (value !== null) {
              return value
            }
          }
        }
      }
      catch (e) {
        console.log('URL parsing failed:', e)
      }
    }
  
    // 尝试从 uni-app 的 getCurrentPages 获取参数（适用于小程序和APP）
    try {
      const pages = getCurrentPages()
      if (pages.length > 0) {
        const currentPage = pages[pages.length - 1]
        const options = (currentPage).options
        if (options && options[str]) {
          return options[str]
        }
  
        // 对于 Vue Router 风格的参数
        const route = (currentPage).$route
        if (route && route.query && route.query[str]) {
          return route.query[str]
        }
      }
    }
    catch (e) {
      console.log('Failed to get parameter from page options')
    }
  
    return null
  }
  ```
