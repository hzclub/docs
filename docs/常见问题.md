## 同步ajax

```js
// 产品导航栏
        $.ajax({
            url: "/api/sortclass.php?parentid=2",
            dataType: "json",
            success: function(res){
                if(res.code == 0){
                    let product = ``;
                    res.data.forEach(item => {
                        $.ajax({
                            url: `/api/index.php?table=product&sortid=${item.id}`,
                            dataType: "json",
                            async: false,	// 同步请求，保证ajax顺序执行，等ajax执行完之后再执行下面代码
                            success: function(res2){
                                product += `<ul>`;
                                product += `<li><a href="product.php">${item.sortname}</a></li>`;
                                res2.data.forEach(item2 => {
                                    product += `<li><a href="productCon.php">${item2.title}</a></li>`;
                                })
                                product += `</ul>`;
                            }
                        })
                    })
                    $(".downmenu.product").html(product);
                }
            }
        })
```

## 拿到依赖注入数组的第一项

```vue
<script setup>
    import { ref, inject, computed } from 'vue'
    import { cates } from '@/utils/key'
    import { useRoute } from 'vue-router'

    let aboutus = ref([])
    aboutus = inject(cates.aboutusCate)

    const route = useRoute()

    const fisrt = computed(()=>{
        let id = route.params.sortid ? route.params.sortid : ""

        return id ? (aboutus.value.find(item => item.id==id) || {}) : aboutus.value[0]
    })
</script>
```
