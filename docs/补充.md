```js
// i18n/index.js
import { createI18n } from "vue-i18n";
import messages from "./messages.js";
const i18n = createI18n({
  legacy: false, // 必须设置为 false 以启用 Composition API 模式
  locale: sessionStorage.getItem('locale') ? sessionStorage.getItem('locale') : "zh",  // 默认语言
  fallbackLocale: "zh", // 回退语言
  globalInjection: true, // 全局注入 $t 方法
  messages, // 语言包
});
export default i18n;

//i18n/messages.js
import zh from "./locale/zh.json"
import en from "./locale/en.json";
export default {
  zh,
  en,
};

// i18n/local/zh.json(插件，js需要默认导出)
{
    "index.index.654021-0": "公告:",	// en.json 键名一样，内容不一样
}

// 页面使用		页面模板使用  {{ $t('键名') }}
import { useI18n } from 'vue-i18n';   // 导入国际化模块
const { locale } = useI18n();		// 设置语言
// 语言切换
function triggerFun(e){
    locale.value = e
    sessionStorage.setItem('locale', e)
}
```

#### JS 国际化

```js
// i18n/index.js
import { createI18n } from "vue-i18n";
import messages from "./messages.js";

const i18n = createI18n({
  legacy: false, // 必须设置为 false 以启用 Composition API 模式
  locale: sessionStorage.getItem("locale")
    ? sessionStorage.getItem("locale")
    : "zh-CN", // 默认语言
  fallbackLocale: "zh-CN", // 回退语言
  globalInjection: true, // 全局注入 $t 方法，主要配置
  messages, // 语言包
});

export default i18n;
```

```js
// 页面
<script setup>
    import { useI18n } from 'vue-i18n';   // 导入国际化模块
	const { t, locale } = useI18n();

    uni.showToast({
        title: t('customize-7'),	// js中使用国际化
        icon: 'none'
    })
</script>
```

## 请求头加密

```js
import CryptoJS from "crypto-js";

// 加密
function setSign(token) {
  const times = Date.now(); // 当前签名有效时间
  const SIGN_SECRET_KEY = import.meta.env.VITE_SIGN_KEY; // 签名密钥(存放在环境变量里面)
  const string = `Bearer ${token}${times}`; // 注意空格
  const sign = CryptoJS.HmacSHA256(string, SIGN_SECRET_KEY).toString(); // 生成签名
  return { sign, times }; // 返回签名和时间戳
}

// 请求拦截器
const requestInterceptor = (config) => {
  const token = useUserStore().token;
  const { sign, times } = setSign(token);
  const Authorization = token ? "Bearer " + token : "";
  // 在这里可以对请求进行处理，例如添加请求头、签名等
  config.header = {
    ...config.header,
    Authorization,
    sign,
    times,
    "Accept-Language": sessionStorage.getItem("locale")
      ? sessionStorage.getItem("locale")
      : "zh-CN", // 请求语言
  };
  return config;
};
```
